## Phase 2 — Refactor `sync_dimensions_from_supabase.py`

ของเดิมทำ DELETE + INSERT ตรง — schema ใหม่ใช้ MERGE pattern ผ่าน SP_SYNC_*

ใหม่: script ทำ 2 steps:
1. **Extract** จาก Supabase → bulk_insert ลง STG_LINE/STG_BATTERY_MODEL/STG_MACHINE (TRUNCATE+LOAD ผ่าน FastAPI)
2. **Transform** เรียก `SP_SYNC_ALL_DIMS` ผ่าน FastAPI → MERGE STG → DIM (preserve surrogate keys)

```python
# Pseudo-code outline
from db_module.db_conn import SupabaseConnector
from app.api.deps import call_sp, bulk_insert  # หรือ HTTP client

def sync():
    run_id = f"manual_dim_sync_{datetime.now().isoformat()}"
    
    # Extract production_line
    with SupabaseConnector().cursor() as cur:
        cur.execute("SELECT line_id, name, area FROM production_line")
        rows = cur.fetchall()
    bulk_insert("STG_LINE", ["line_id","name","area"], rows, 
                truncate=True, pipeline_run_id=run_id)
    
    # Extract battery_model
    with SupabaseConnector().cursor() as cur:
        cur.execute("""
            SELECT model_id, model_code, name, spec_plate_count, spec_weight_kg,
                   spec_terminal_type, dim_length_mm, dim_width_mm, dim_height_mm,
                   is_active 
              FROM battery_model
        """)
        rows = cur.fetchall()
    bulk_insert("STG_BATTERY_MODEL", [...], rows, truncate=True, pipeline_run_id=run_id)
    
    # Extract machine
    with SupabaseConnector().cursor() as cur:
        cur.execute("""
            SELECT machine_id, line_id, machine_code, machine_type,
                   sequence_position, install_date, is_active
              FROM machine
        """)
        rows = cur.fetchall()
    bulk_insert("STG_MACHINE", [...], rows, truncate=True, pipeline_run_id=run_id)
    
    # Transform: STG → DIM (preserves surrogate keys via MERGE)
    call_sp("SP_SYNC_ALL_DIMS")
```

### Updated SQL extracts

```python
SQL_PRODUCTION_BATCH = """
SELECT b.batch_id, b.order_id, b.line_id,
       o.model_id,                          -- renamed from product_id
       b.qty_planned, b.qty_out, 
       b.start_time, b.end_time,
       o.planned_start AS order_planned_start,
       o.planned_end   AS order_planned_end,
       (SELECT SUM(qty_planned) FROM production_batch 
         WHERE order_id = b.order_id) AS order_total_qty
  FROM production_batch b
  JOIN production_order o ON b.order_id = o.order_id
 WHERE b.end_time IS NOT NULL
   AND b.end_time >= %s AND b.end_time < %s
"""

SQL_QC_RECORD = """
SELECT qc_id, batch_id, 
       qty_inspected,                       -- renamed from qty_sampled
       qty_passed, qty_failed, inspected_at
  FROM qc_record
 WHERE inspected_at >= %s AND inspected_at < %s
"""

SQL_QC_DEFECT = """
SELECT qd.qc_id, qd.defect_code, qd.qty_affected
  FROM qc_defect qd
  JOIN qc_record qc ON qc.qc_id = qd.qc_id
 WHERE qc.inspected_at >= %s AND qc.inspected_at < %s
"""

SQL_DOWNTIME_EVENT = """
SELECT de.event_id, 
       de.machine_id,                       -- Supabase business key
       m.machine_code,                      -- M01/M02/M03 lookup
       m.line_id,
       de.batch_id, de.reason_code,
       er.is_planned,                       -- from event_reason JOIN
       de.start_ts, de.end_ts, de.duration_min
  FROM downtime_event de
  JOIN machine m ON m.machine_id = de.machine_id
  JOIN event_reason er ON er.reason_code = de.reason_code
 WHERE de.end_ts IS NOT NULL                -- only closed events
   AND de.end_ts >= %s AND de.end_ts < %s
"""
```
