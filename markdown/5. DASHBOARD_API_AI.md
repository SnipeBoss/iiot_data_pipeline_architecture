```python
def oee_trend_chart(df: pd.DataFrame) -> go.Figure:
    """OEE trend with components (availability/performance/quality).
    
    Expected columns: full_date, oee_pct, availability_pct, 
                      performance_pct, quality_pct
    """
    fig = go.Figure()
    fig.add_trace(go.Scatter(x=df['full_date'], y=df['oee_pct'],
                             name='OEE', mode='lines+markers',
                             line=dict(color='#0F6E56', width=3)))
    fig.add_trace(go.Scatter(x=df['full_date'], y=df['availability_pct'],
                             name='Availability', mode='lines',
                             line=dict(color='#534AB7', dash='dash')))
    fig.add_trace(go.Scatter(x=df['full_date'], y=df['performance_pct'],
                             name='Performance', mode='lines',
                             line=dict(color='#185FA5', dash='dot')))
    fig.add_trace(go.Scatter(x=df['full_date'], y=df['quality_pct'],
                             name='Quality', mode='lines',
                             line=dict(color='#BA7517', dash='dashdot')))
    fig.update_layout(
        height=350, hovermode='x unified',
        yaxis=dict(title='%', range=[0, 105]),
        xaxis=dict(title=None),
        margin=dict(t=20, b=40, l=40, r=20),
    )
    return fig


def defect_pareto_chart(df: pd.DataFrame) -> go.Figure:
    """Pareto chart: bars sorted desc + cumulative % line.
    
    Expected columns: defect_code, qty_affected, pct_of_total
    """
    df = df.sort_values('qty_affected', ascending=False).head(10)
    df['cum_pct'] = df['pct_of_total'].cumsum()
    
    fig = go.Figure()
    fig.add_trace(go.Bar(x=df['defect_code'], y=df['qty_affected'],
                         name='Qty', marker_color='#D85A30'))
    fig.add_trace(go.Scatter(x=df['defect_code'], y=df['cum_pct'],
                             name='Cumulative %', mode='lines+markers',
                             yaxis='y2', line=dict(color='#534AB7')))
    fig.update_layout(
        height=350,
        yaxis=dict(title='Quantity'),
        yaxis2=dict(title='Cumulative %', overlaying='y', side='right',
                    range=[0, 105]),
        margin=dict(t=20, b=80, l=40, r=40),

def forecast_chart(historical: pd.DataFrame, forecast: pd.DataFrame, 
                   threshold: float | None = None) -> go.Figure:
    """Prophet forecast with confidence band + threshold line.
    
    historical columns: ds, y
    forecast columns: ds, yhat, yhat_lower, yhat_upper
    """
    fig = go.Figure()
    
    # Historical (blue solid)
    fig.add_trace(go.Scatter(x=historical['ds'], y=historical['y'],
                             name='Historical', mode='lines',
                             line=dict(color='#185FA5', width=2)))
    
    # Forecast (orange dashed)
    fig.add_trace(go.Scatter(x=forecast['ds'], y=forecast['yhat'],
                             name='Forecast', mode='lines',
                             line=dict(color='#BA7517', width=2, dash='dash')))
    
    # Confidence band (yellow shaded)
    fig.add_trace(go.Scatter(
        x=list(forecast['ds']) + list(forecast['ds'][::-1]),
        y=list(forecast['yhat_upper']) + list(forecast['yhat_lower'][::-1]),
        fill='toself', fillcolor='rgba(250, 199, 117, 0.3)',
        line=dict(color='rgba(255,255,255,0)'),
        name='Confidence band', hoverinfo='skip'
    ))
    
    # Threshold (red dashed)
    if threshold is not None:
        fig.add_hline(y=threshold, line_dash='dash', line_color='#E24B4A',
                      annotation_text=f'threshold {threshold}', 
                      annotation_position='top right')
    
    # "Now" vertical line at boundary
    if not historical.empty:
        now = historical['ds'].max()
        fig.add_vline(x=now, line_dash='dash', line_color='#888780',
                      annotation_text='now', annotation_position='top left')
    
    fig.update_layout(
        height=400, hovermode='x unified',
        xaxis=dict(title=None),
        yaxis=dict(title='Value'),
        margin=dict(t=20, b=40, l=40, r=20),
    )
    return fig

def slippage_histogram(df: pd.DataFrame) -> go.Figure:
    """Histogram of slippage_min with color zones.
    
    Expected columns: slippage_min
    """
    fig = go.Figure()
    fig.add_trace(go.Histogram(
        x=df['slippage_min'], nbinsx=20,
        marker=dict(color='#1D9E75'),
    ))
    fig.add_vline(x=5, line_dash='dash', line_color='#888780',
                  annotation_text='on-time threshold')
    fig.add_vline(x=15, line_dash='dash', line_color='#E24B4A',
                  annotation_text='late threshold')
    fig.update_layout(
        height=300,
        xaxis=dict(title='Slippage (minutes)'),
        yaxis=dict(title='Batch count'),
        margin=dict(t=20, b=40, l=40, r=20),
    )
    return fig


def slippage_trend(df: pd.DataFrame) -> go.Figure:
    """Avg slippage per date.
    
    Expected: slippage_min + a date column (full_date or planned_start)
    """
    if 'full_date' in df.columns:
        date_col = 'full_date'
    else:
        df['__date'] = pd.to_datetime(df['order_planned_start']).dt.date
        date_col = '__date'
    
    daily = df.groupby(date_col)['slippage_min'].mean().reset_index()
    fig = go.Figure()
    fig.add_trace(go.Scatter(x=daily[date_col], y=daily['slippage_min'],
                             mode='lines+markers',
                             line=dict(color='#0F6E56', width=2)))
    fig.add_hline(y=0, line_dash='dot', line_color='#888780')
    fig.update_layout(
        height=300,
        xaxis=dict(title=None),
        yaxis=dict(title='Avg slippage (min)'),
        margin=dict(t=20, b=40, l=40, r=20),
    )
    return fig


def batch_gantt(df: pd.DataFrame) -> go.Figure:
    """Gantt chart for batches in selected order (planned vs actual).
    
    Expected columns: batch_src_id, batch_planned_start, batch_planned_end,
                      start_time, end_time
    """
    fig = go.Figure()
    
    for _, row in df.iterrows():
        bid = f"batch {row['batch_src_id']}"
        # Planned (gray)
        if row.get('batch_planned_start') and row.get('batch_planned_end'):
            fig.add_trace(go.Scatter(
                x=[row['batch_planned_start'], row['batch_planned_end']],
                y=[bid, bid], mode='lines',
                line=dict(color='#B4B2A9', width=12),
                showlegend=False, hovertext=f'planned: {bid}'
            ))
        # Actual (green if on-time, red if late)
        slip = row.get('slippage_min', 0) or 0
        color = '#E24B4A' if slip > 15 else '#1D9E75'
        if row.get('start_time') and row.get('end_time'):
            fig.add_trace(go.Scatter(
                x=[row['start_time'], row['end_time']],
                y=[bid, bid], mode='lines',
                line=dict(color=color, width=8),
                showlegend=False, hovertext=f'actual: {bid} (slip {slip:.1f} min)'
            ))
    
    fig.update_layout(
        height=max(300, 30 * len(df) + 100),
        xaxis=dict(title=None, type='date'),
        yaxis=dict(title=None, autorange='reversed'),
        margin=dict(t=20, b=40, l=80, r=20),
    )
    return fig
```

```python
"""Reusable KPI card components."""
import streamlit as st


def kpi_card(label: str, value: str | float, delta: str | None = None,
             help: str | None = None) -> None:
    """Wraps st.metric with consistent formatting.
    
    Args:
        label: card title (e.g. "OEE")
        value: main number (formatted string or numeric)
        delta: optional delta text (e.g. "+2.3%")
        help: optional tooltip
    """
    st.metric(label=label, value=value, delta=delta, help=help)


def kpi_row(items: list[dict]) -> None:
    """Render a row of KPI cards in equal columns.
    
    Args:
        items: list of dicts with keys: label, value, delta?, help?
    
    Example:
        kpi_row([
            {"label": "OEE", "value": "78%", "delta": "+2%"},
            {"label": "Availability", "value": "92%"},
            {"label": "Performance", "value": "88%"},
            {"label": "Quality", "value": "96%"},
        ])
    """
    cols = st.columns(len(items))
    for col, item in zip(cols, items):
        with col:
            kpi_card(**item)


def status_badge(text: str, status: str = "info") -> str:
    """Markdown badge with color.
    
    status: 'success' (green), 'warning' (amber), 'danger' (red), 'info' (blue)
    """
    colors = {
        'success': ('#1D9E75', 'white'),
        'warning': ('#BA7517', 'white'),
        'danger': ('#E24B4A', 'white'),
        'info': ('#185FA5', 'white'),
    }
    bg, fg = colors.get(status, colors['info'])
    return (f'<span style="background:{bg};color:{fg};padding:2px 10px;'
            f'border-radius:11px;font-size:12px;font-weight:500;">'
            f'{text}</span>')
```

### Component 4: `components/prophet_trainer.py` (Background training)

```python
"""Prophet background training + inference helper."""
import os
import pickle
import threading
import datetime
from pathlib import Path

import pandas as pd

CACHE_DIR = Path(__file__).parent.parent / "cache" / "prophet_models"
CACHE_DIR.mkdir(parents=True, exist_ok=True)

# Per-process training state
_training_status: dict[str, str] = {}  # key: f"{machine}_{metric}"


def model_path(machine: str, metric: str) -> Path:
    """Returns path to saved .pkl file."""
    return CACHE_DIR / f"{machine}_{metric}.pkl"


def model_status(machine: str, metric: str) -> dict:
    """Inspect cached model file.
    
    Returns dict with keys: exists, last_trained, status_text
    """
    key = f"{machine}_{metric}"
    
    # Currently training?
    if _training_status.get(key) == "training":
        return {"exists": False, "last_trained": None,
                "status_text": "Training in progress..."}
    
    # File exists?
    p = model_path(machine, metric)
    if not p.exists():
        return {"exists": False, "last_trained": None,
                "status_text": "Not trained yet — click Train"}
    
    mtime = datetime.datetime.fromtimestamp(p.stat().st_mtime)
    return {"exists": True, "last_trained": mtime,
            "status_text": f"Trained at {mtime.strftime('%Y-%m-%d %H:%M')}"}


def _train_in_thread(machine: str, metric: str, history_df: pd.DataFrame) -> None:
    """Internal: fit Prophet + save .pkl."""
    from prophet import Prophet
    
    key = f"{machine}_{metric}"
    _training_status[key] = "training"
    
    try:
        model = Prophet(daily_seasonality=True, weekly_seasonality=True,
                        interval_width=0.95)
        model.fit(history_df)
        
        with open(model_path(machine, metric), "wb") as f:
            pickle.dump(model, f)
        
        _training_status[key] = "ready"
    except Exception as e:
        _training_status[key] = f"error: {e}"


def trigger_training(machine: str, metric: str, history_df: pd.DataFrame) -> None:
    """Start Prophet training in background thread.
    
    history_df must have columns: ds (datetime), y (numeric)
    """
    if history_df.empty or len(history_df) < 30:
        raise ValueError("Need at least 30 historical points to train")
    
    thread = threading.Thread(
        target=_train_in_thread,
        args=(machine, metric, history_df.copy()),
        daemon=True,
    )
    thread.start()


def predict(machine: str, metric: str, hours: int) -> pd.DataFrame | None:
    """Load .pkl + forecast next N hours.
    
    Returns df with columns ds, yhat, yhat_lower, yhat_upper or None if no model.
    """
    p = model_path(machine, metric)
    if not p.exists():
        return None
    
    with open(p, "rb") as f:
        model = pickle.load(f)
    
    # 15-min steps
    periods = hours * 4
    future = model.make_future_dataframe(periods=periods, freq='15min',
                                         include_history=False)
    forecast = model.predict(future)
    return forecast[['ds', 'yhat', 'yhat_lower', 'yhat_upper']]
```

```python
"""Page 2 — Sensor Forecast (Prophet).

Components:
  A: Model status card
  B: Forecast chart (historical + forecast + confidence band)
  + (NEW) Compare 2 metrics side-by-side panel
"""
import datetime
import streamlit as st
import pandas as pd

from components import api_client as api
from components.filters import filter_row_forecast
from components.cards import status_badge
from components.charts import forecast_chart, metric_comparison_chart
from components import prophet_trainer as pt


st.set_page_config(page_title="Sensor Forecast", layout="wide")
st.title("🔮 Sensor Forecast (Prophet)")

# ─── Fetch metrics list for filter dropdowns ──────────────
metrics_list = api.get("/api/sensor/available-metrics")
if not metrics_list:
    st.error("No sensor metrics defined in DIM_METRIC")
    st.stop()

# ─── Filters ──────────────────────────────────────────────
filters = filter_row_forecast(metrics_list)
machine = filters["machine"]
metric = filters["metric"]
horizon_hours = int(filters["horizon"].split()[0])
train_clicked = filters["train_clicked"]

# Lookup threshold for this metric
metric_meta = next((m for m in metrics_list if m['metric_name'] == metric), {})
threshold = metric_meta.get('critical_threshold')

# ─── Fetch historical (last 7 days) ──────────────────────
end_date = datetime.date.today()
all_history = []
for i in range(7):
    d = end_date - datetime.timedelta(days=i)
    try:
        chunk = api.get("/api/sensor/by-machine-15min",
                        {"date": d.isoformat(), "metric": metric})
        rows = chunk.get("rows", [])
        # Filter for selected machine
        machine_rows = [r for r in rows if r.get('machine_code') == machine]
        all_history.extend(machine_rows)
    except Exception:
        continue

history_df = pd.DataFrame(all_history)

if not history_df.empty:
    # Prophet expects ds, y
    history_df = history_df.rename(columns={
        'window_start': 'ds',
        'avg_value': 'y',
    })
    history_df['ds'] = pd.to_datetime(history_df['ds'])
    history_df = history_df.sort_values('ds').reset_index(drop=True)

# ─── Train trigger ────────────────────────────────────────
if train_clicked:
    if history_df.empty or len(history_df) < 30:
        st.error(f"Need ≥30 historical points; got {len(history_df)}")
    else:
        try:
            pt.trigger_training(machine, metric, history_df[['ds', 'y']])
            st.success(f"Training started for {machine} / {metric}. Refresh in 30s.")
        except Exception as e:
            st.error(f"Training failed: {e}")

# ─── Component A: Model status card ───────────────────────
status = pt.model_status(machine, metric)

with st.container(border=True):
    cols = st.columns([3, 1])
    with cols[0]:
        if status["exists"]:
            st.markdown(f"**{machine} / {metric}**  \n{status['status_text']}")
        else:
            st.markdown(f"**{machine} / {metric}**  \n_{status['status_text']}_")
    with cols[1]:
        if status["exists"]:
            st.markdown(status_badge("Ready", "success"), unsafe_allow_html=True)
        elif "Training" in status['status_text']:
            st.markdown(status_badge("Training", "warning"), unsafe_allow_html=True)
        else:
            st.markdown(status_badge("Not ready", "danger"), unsafe_allow_html=True)

st.divider()

# ─── Component B: Forecast chart ──────────────────────────
st.subheader("Forecast")

if not status["exists"]:
    st.info("Train a model first to see the forecast")
else:
    forecast_df = pt.predict(machine, metric, horizon_hours)
    if forecast_df is None or forecast_df.empty:
        st.warning("Could not generate forecast")
    else:
        st.plotly_chart(
            forecast_chart(history_df[['ds', 'y']], forecast_df, threshold),
            use_container_width=True,
        )

st.divider()

# ─── Compare 2 metrics side-by-side (NEW) ─────────────────
st.subheader("Compare 2 metrics")

with st.expander("Metric comparison", expanded=False):
    machine_metrics = [m['metric_name'] for m in metrics_list 
                       if m.get('machine_code') == machine or m.get('machine_code') is None]
    
    cols = st.columns(2)
    with cols[0]:
        cmp_a = st.selectbox("Metric A", machine_metrics, key="cmp_a")
    with cols[1]:
        cmp_b = st.selectbox("Metric B", machine_metrics, 
                             index=min(1, len(machine_metrics)-1), key="cmp_b")
    
    if cmp_a == cmp_b:
        st.warning("Pick 2 different metrics")
    else:
        # Fetch both
        all_a, all_b = [], []
        for i in range(7):
            d = end_date - datetime.timedelta(days=i)
            try:
                chunk_a = api.get("/api/sensor/by-machine-15min",
                                  {"date": d.isoformat(), "metric": cmp_a})
                chunk_b = api.get("/api/sensor/by-machine-15min",
                                  {"date": d.isoformat(), "metric": cmp_b})
                all_a.extend(r for r in chunk_a.get("rows", []) 
                             if r.get('machine_code') == machine)
                all_b.extend(r for r in chunk_b.get("rows", []) 
                             if r.get('machine_code') == machine)
            except Exception:
                continue
        
        if all_a and all_b:
            df_a = pd.DataFrame(all_a).rename(columns={'window_start': 'ds', 'avg_value': 'y'})
            df_b = pd.DataFrame(all_b).rename(columns={'window_start': 'ds', 'avg_value': 'y'})
            df_a['ds'] = pd.to_datetime(df_a['ds'])
            df_b['ds'] = pd.to_datetime(df_b['ds'])
            st.plotly_chart(
                metric_comparison_chart(df_a, df_b, cmp_a, cmp_b),
                use_container_width=True,
            )
        else:
            st.info("No data to compare")
```
